rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ══════════════════════════════════════════════════════════════
    // Phase 5 hardening (11f). The model, honestly stated:
    // - Polls are public: anyone with the link reads and votes, no
    //   account required. Rules cannot iterate arrays, so per-vote
    //   validation inside the dates/games arrays is impossible; the
    //   caps below plus Firebase App Check are the anti-abuse layer
    //   for anonymous traffic.
    // - A poll WITHOUT ownerUid (anonymous creator) keeps client-side
    //   token gating for management: the token sits in the public
    //   document, so rules could never treat it as a secret anyway.
    // - A poll WITH ownerUid is rules-protected: management fields
    //   (title, closed, deadline, finalized date, capacity), resizing
    //   the dates list and shrinking games require the owner account.
    //   Everyone else can only vote/comment (dates keep their count),
    //   suggest at most one game per write, or claim their own votes.
    // - Deletion always requires the signed-in owner.
    // ══════════════════════════════════════════════════════════════

    function signedIn() { return request.auth != null; }

    match /polls/{pollId} {
      allow read: if true;

      // Shape + bounds. Optional-field checks are presence-guarded so
      // an existing poll that predates a field (e.g. the very first
      // test polls, created before `closed`/`creatorToken` existed)
      // can still be voted on. The create rule below additionally
      // REQUIRES creatorToken and closed for new polls.
      function validShape(d) {
        return d.keys().hasOnly(['id', 'title', 'createdAt', 'dates', 'creatorToken', 'closed', 'deadline', 'finalizedDateId', 'minPlayers', 'maxPlayers', 'games', 'ownerUid']) &&
          d.id == pollId &&
          d.title is string && d.title.size() > 0 && d.title.size() <= 100 &&
          d.createdAt is timestamp &&
          (!('creatorToken' in d) || (d.creatorToken is string && d.creatorToken.size() >= 8 && d.creatorToken.size() <= 64)) &&
          (!('closed' in d) || d.closed is bool) &&
          d.dates is list && d.dates.size() > 0 && d.dates.size() <= 92 &&
          (!('games' in d) || (d.games is list && d.games.size() <= 30)) &&
          (!('deadline' in d) || d.deadline is timestamp) &&
          (!('finalizedDateId' in d) || (d.finalizedDateId is string && d.finalizedDateId.size() <= 40)) &&
          (!('minPlayers' in d) || (d.minPlayers is int && d.minPlayers >= 1 && d.minPlayers <= 99)) &&
          (!('maxPlayers' in d) || (d.maxPlayers is int && d.maxPlayers >= 1 && d.maxPlayers <= 99)) &&
          (!('ownerUid' in d) || (d.ownerUid is string && d.ownerUid.size() > 0 && d.ownerUid.size() <= 128));
      }

      allow create: if
        validShape(request.resource.data) &&
        'creatorToken' in request.resource.data &&
        'closed' in request.resource.data &&
        request.resource.data.closed == false &&
        !('games' in request.resource.data) &&
        !('finalizedDateId' in request.resource.data) &&
        (!('ownerUid' in request.resource.data) ||
          (signedIn() && request.resource.data.ownerUid == request.auth.uid));

      // Identity fields never change; ownerUid can be set exactly
      // once, only by the signed-in account it names (the claim)
      function identityKept() {
        return request.resource.data.id == resource.data.id &&
          request.resource.data.createdAt == resource.data.createdAt &&
          request.resource.data.get('creatorToken', '') == resource.data.get('creatorToken', '') &&
          (
            (('ownerUid' in request.resource.data) == ('ownerUid' in resource.data) &&
              (!('ownerUid' in resource.data) || request.resource.data.ownerUid == resource.data.ownerUid)) ||
            (!('ownerUid' in resource.data) && signedIn() &&
              request.resource.data.ownerUid == request.auth.uid)
          );
      }

      function isOwner() {
        return ('ownerUid' in resource.data) && signedIn() &&
          request.auth.uid == resource.data.ownerUid;
      }

      // What a non-owner may do to an owned poll: change nothing at
      // the management level. Same optional-field presence and value,
      // same number of dates; games may grow by at most one entry
      // (a suggestion) and never shrink.
      function managementUntouched() {
        return request.resource.data.title == resource.data.title &&
          request.resource.data.closed == resource.data.closed &&
          (('deadline' in request.resource.data) == ('deadline' in resource.data) &&
            (!('deadline' in resource.data) || request.resource.data.deadline == resource.data.deadline)) &&
          (('finalizedDateId' in request.resource.data) == ('finalizedDateId' in resource.data) &&
            (!('finalizedDateId' in resource.data) || request.resource.data.finalizedDateId == resource.data.finalizedDateId)) &&
          (('minPlayers' in request.resource.data) == ('minPlayers' in resource.data) &&
            (!('minPlayers' in resource.data) || request.resource.data.minPlayers == resource.data.minPlayers)) &&
          (('maxPlayers' in request.resource.data) == ('maxPlayers' in resource.data) &&
            (!('maxPlayers' in resource.data) || request.resource.data.maxPlayers == resource.data.maxPlayers)) &&
          request.resource.data.dates.size() == resource.data.dates.size() &&
          request.resource.data.get('games', []).size() >= resource.data.get('games', []).size() &&
          request.resource.data.get('games', []).size() <= resource.data.get('games', []).size() + 1;
      }

      allow update: if
        validShape(request.resource.data) &&
        identityKept() &&
        (
          isOwner() ||
          !('ownerUid' in resource.data) ||
          managementUntouched()
        );

      // Only the signed-in owner can delete; unclaimed anonymous
      // polls are undeletable (their creator signs in and ownership
      // is claimed on visit)
      allow delete: if isOwner();
    }

    // Per-account document holding the user's synced poll list.
    // Only the signed-in owner touches it; a single `polls` map,
    // capped comfortably above the client's 50-entry limit.
    match /users/{userId} {
      allow read: if signedIn() && request.auth.uid == userId;
      allow create, update: if
        signedIn() &&
        request.auth.uid == userId &&
        request.resource.data.keys().hasOnly(['polls']) &&
        request.resource.data.polls is map &&
        request.resource.data.polls.size() <= 60;
      allow delete: if false;
    }
  }
}

// Notes:
// - Per-element validation of votes/comments/games entries is not
//   expressible in rules (no array iteration); document size is
//   bounded by Firestore's 1 MB limit and the caps above. App Check
//   (enforced after 11f's rollout completes) is the abuse gate for
//   anonymous writes.
// - The Vercel expiry cron uses the Admin SDK and bypasses these
//   rules and App Check by design; it authenticates with CRON_SECRET.
