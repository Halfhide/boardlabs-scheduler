rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /polls/{pollId} {
      allow read: if true;

      // Anyone can create a poll, but only with the expected shape
      // and within size limits. A poll created while signed in may
      // carry its owner's uid, which must be the requester's own.
      allow create: if
        request.resource.data.keys().hasOnly(['id', 'title', 'createdAt', 'dates', 'creatorToken', 'closed', 'deadline', 'finalizedDateId', 'minPlayers', 'maxPlayers', 'ownerUid']) &&
        (!('ownerUid' in request.resource.data) || (request.auth != null && request.resource.data.ownerUid == request.auth.uid)) &&
        (!('deadline' in request.resource.data) || request.resource.data.deadline is timestamp) &&
        (!('finalizedDateId' in request.resource.data) || request.resource.data.finalizedDateId is string) &&
        (!('minPlayers' in request.resource.data) || (request.resource.data.minPlayers is int && request.resource.data.minPlayers >= 1 && request.resource.data.minPlayers <= 99)) &&
        (!('maxPlayers' in request.resource.data) || (request.resource.data.maxPlayers is int && request.resource.data.maxPlayers >= 1 && request.resource.data.maxPlayers <= 99)) &&
        request.resource.data.id == pollId &&
        request.resource.data.title is string &&
        request.resource.data.title.size() > 0 &&
        request.resource.data.title.size() <= 100 &&
        request.resource.data.createdAt is timestamp &&
        request.resource.data.creatorToken is string &&
        request.resource.data.creatorToken.size() >= 8 &&
        request.resource.data.creatorToken.size() <= 64 &&
        request.resource.data.closed == false &&
        request.resource.data.dates is list &&
        request.resource.data.dates.size() > 0 &&
        request.resource.data.dates.size() <= 92;

      // Updates cover votes/comments (dates array rewrites) and
      // creator actions (rename, add/remove dates, close/reopen).
      // The poll's id, creation time and creator token are immutable
      // and every field stays within its bounds. The owner uid can be
      // set exactly once, only by the signed-in account it names (the
      // ownership claim); once set it never changes. Anonymous
      // creator identity stays client-side until the full rules
      // rewrite (11f): the document is publicly readable, so the
      // token cannot act as a server-side secret anyway.
      allow update: if
        request.resource.data.keys().hasOnly(['id', 'title', 'createdAt', 'dates', 'creatorToken', 'closed', 'deadline', 'finalizedDateId', 'minPlayers', 'maxPlayers', 'games', 'ownerUid']) &&
        (
          request.resource.data.get('ownerUid', '') == resource.data.get('ownerUid', '') ||
          (resource.data.get('ownerUid', '') == '' && request.auth != null && request.resource.data.ownerUid == request.auth.uid)
        ) &&
        (!('games' in request.resource.data) || (request.resource.data.games is list && request.resource.data.games.size() <= 30)) &&
        (!('deadline' in request.resource.data) || request.resource.data.deadline is timestamp) &&
        (!('finalizedDateId' in request.resource.data) || request.resource.data.finalizedDateId is string) &&
        (!('minPlayers' in request.resource.data) || (request.resource.data.minPlayers is int && request.resource.data.minPlayers >= 1 && request.resource.data.minPlayers <= 99)) &&
        (!('maxPlayers' in request.resource.data) || (request.resource.data.maxPlayers is int && request.resource.data.maxPlayers >= 1 && request.resource.data.maxPlayers <= 99)) &&
        request.resource.data.id == resource.data.id &&
        request.resource.data.createdAt == resource.data.createdAt &&
        request.resource.data.get('creatorToken', '') == resource.data.get('creatorToken', '') &&
        request.resource.data.title is string &&
        request.resource.data.title.size() > 0 &&
        request.resource.data.title.size() <= 100 &&
        request.resource.data.get('closed', false) is bool &&
        request.resource.data.dates is list &&
        request.resource.data.dates.size() > 0 &&
        request.resource.data.dates.size() <= 92;

      // Only the signed-in owner can delete a poll. Polls without an
      // ownerUid (anonymous, never claimed) cannot be deleted; their
      // creator signs in first and ownership is claimed on visit.
      allow delete: if
        request.auth != null &&
        resource.data.get('ownerUid', '') == request.auth.uid;
    }

    // Per-account document holding the user's synced poll list.
    // Only the signed-in owner can read or write it; the shape is a
    // single `polls` map (entries validated in the 11f rules pass).
    match /users/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow create, update: if
        request.auth != null &&
        request.auth.uid == userId &&
        request.resource.data.keys().hasOnly(['polls']) &&
        request.resource.data.polls is map;
      allow delete: if false;
    }
  }
}

// Note: without authentication these rules cannot stop a determined
// visitor from rewriting votes or poll settings of a poll whose link
// they have. They do prevent deletion by anyone but the signed-in
// owner, malformed or oversized documents, forging a poll's identity
// fields, and touching any other collection. This app is intended
// for friendly groups, not adversarial environments.
