rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /polls/{pollId} {
      allow read: if true;

      // Anyone can create a poll, but only with the expected shape
      // and within size limits
      allow create: if
        request.resource.data.keys().hasOnly(['id', 'title', 'createdAt', 'dates', 'creatorToken', 'closed', 'deadline']) &&
        (!('deadline' in request.resource.data) || request.resource.data.deadline is timestamp) &&
        request.resource.data.id == pollId &&
        request.resource.data.title is string &&
        request.resource.data.title.size() > 0 &&
        request.resource.data.title.size() <= 100 &&
        request.resource.data.createdAt is timestamp &&
        request.resource.data.creatorToken is string &&
        request.resource.data.creatorToken.size() >= 8 &&
        request.resource.data.creatorToken.size() <= 64 &&
        request.resource.data.closed == false &&
        request.resource.data.dates is list &&
        request.resource.data.dates.size() > 0 &&
        request.resource.data.dates.size() <= 92;

      // Updates cover votes/comments (dates array rewrites) and
      // creator actions (rename, add/remove dates, close/reopen).
      // The poll's id, creation time and creator token are immutable
      // and every field stays within its bounds. Creator identity is
      // enforced client-side until real auth (roadmap phase 5): the
      // document is publicly readable, so the token cannot act as a
      // server-side secret anyway.
      allow update: if
        request.resource.data.keys().hasOnly(['id', 'title', 'createdAt', 'dates', 'creatorToken', 'closed', 'deadline']) &&
        (!('deadline' in request.resource.data) || request.resource.data.deadline is timestamp) &&
        request.resource.data.id == resource.data.id &&
        request.resource.data.createdAt == resource.data.createdAt &&
        request.resource.data.get('creatorToken', '') == resource.data.get('creatorToken', '') &&
        request.resource.data.title is string &&
        request.resource.data.title.size() > 0 &&
        request.resource.data.title.size() <= 100 &&
        request.resource.data.get('closed', false) is bool &&
        request.resource.data.dates is list &&
        request.resource.data.dates.size() > 0 &&
        request.resource.data.dates.size() <= 92;

      allow delete: if false;
    }
  }
}

// Note: without authentication these rules cannot stop a determined
// visitor from rewriting votes or poll settings of a poll whose link
// they have. They do prevent deletion, malformed or oversized
// documents, forging a poll's identity fields, and touching any other
// collection. This app is intended for friendly groups, not
// adversarial environments.
